2. flex-wrap

flex-wrap: wrap; → إذا كان عرض الأبناء أكبر من الأب، ينزلوا لسطر جديد

flex-wrap: nowrap; → يضغطهم في سطر واحد (الافتراضي)

flex-wrap: wrap-reverse; → ينزلوا لسطر جديد لكن باتجاه معاكس


   .flex-container {
    display: flex;
    flex-wrap: wrap;
}

3. justify-content

justify-content: center; → يوسطن العناصر أفقيًا داخل الحاوية

justify-content: space-between; → يوزع المسافات بين العناصر (أول وآخر عنصر على الحواف)

justify-content: space-around; → مسافات متساوية حول العناصر

justify-content: space-evenly; → مسافات متساوية بين العناصر وحولها

justify-content: flex-start; → يلصقهم بالبداية (الافتراضي)

justify-content: flex-end; → يلصقهم بالنهاية


.flex-container {
    display: flex;
    justify-content: center; /* أو space-between */
}

4. align-items

align-items: center; → يوسطن العناصر عموديًا داخل الحاوية

align-items: flex-start; → يلصقهم بالبداية (فوق)

align-items: flex-end; → يلصقهم بالنهاية (تحت)

align-items: stretch; → يمد العناصر لتملأ الارتفاع (الافتراضي)


  .flex-container {
    display: flex;
    align-items: center;
}

5. flex-direction

flex-direction: row; → العناصر تصطف أفقيًا من اليسار لليمين (الافتراضي)

flex-direction: row-reverse; → أفقيًا من اليمين لليسار

flex-direction: column; → العناصر تصطف عموديًا من فوق لتحت

flex-direction: column-reverse; → عموديًا من تحت لفوق


  .flex-container {
    display: flex;
    flex-direction: column;
}

7. vh (viewport height)

height: 100vh; → ارتفاع العنصر 100% من ارتفاع شاشة المتصفح

1vh = 1% من ارتفاع الشاشة

الفرق بين % و vh:

  • % → نسبة من العنصر الأب
  • vh → نسبة من ارتفاع الشاشة (ثابت)

.element {
    height: 100vh;
}

position: sticky;

المعنى: العنصر يبقى عادي لحد ما يوصل نقطة معينة ثم يثبت

الاستخدام: للهيدر أو القوائم التي تلتصق بالأعلى عند التمرير


sticky-element {
    position: sticky;
    top: 0;
}

::after (العنصر الوهمي)

المعنى: يضيف محتوى وهمي بعد محتوى العنصر الأصلي

الاستخدام: للإضافات الزخرفية مثل الأيقونات أو الخطوط


.box::after {
    content: " ";
    position: absolute;
    width: 70%;
    height: 10px;
    background-color: blueviolet;
    bottom: 0;
    left: 0;
}

transition

المعنى: يضيف حركة سلسة عند تغيير الخاصية

الاستخدام: لجعل التغيرات تبدو احترافية وسلسة


.box:hover::after {
    width: 100%;
    height: 100%;
    transition: 2s;
}
