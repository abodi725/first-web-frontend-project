شرح خصائص CSS

1. width & height

width: 60%; → عرض العنصر 60% من العنصر الأب

height: 7rem; → ارتفاع العنصر 7 × 16px = 112px

                    
 /* width & height */
.element {
    width: 60%;
    height: 7rem;
}

2. background-color & color

background-color: rgb(81, 194, 247); → لون خلفية أزرق فاتح

color: brown; → لون النص بني

                    

/* background-color & color */
.element {
    background-color: rgb(81, 194, 247);
    color: brown;
}

3. opacity

opacity: 0.5; → شفافية العنصر (0=شفاف, 1=غير شفاف)

                    
/* opacity */
.element {
    opacity: 0.5;
}

4. display vs visibility

display: none; → يخفي العنصر ويزيل مكانه بالكامل

visibility: hidden; → يخفي العنصر لكن يبقي مكانه فارغ

                    
/* display: none - يخفي ويزيل المكان */
.hide-display {
    display: none;
}

/* visibility: hidden - يخفي لكن يبقي المكان */
.hide-visibility {
    visibility: hidden;
}

5. font-size & font-family

font-size: 2rem; → حجم الخط 2 × 16px = 32px

font-family: Arial, sans-serif; → نوع الخط

                    
/* font-size & font-family */
.element {
    font-size: 2rem;
    font-family: Arial, sans-serif;
}

6. text-align

text-align: center; → توسيط النص أفقيًا

                    
   /* text-align */
.element {
    text-align: center;
}

7. border

border: solid blue 4px; → حد: نوعه صلب، لونه أزرق، سماكته 4px

                    
 /* border */
.element {
    border: solid blue 4px;
}

8. padding (المسافة الداخلية)

padding: 1rem 2rem 3rem 4rem; → أعلى | يمين | أسفل | يسار

1rem = 16px → أعلى 16px، يمين 32px، أسفل 48px، يسار 64px

                    
/* padding */
.element {
    padding: 1rem 2rem 3rem 4rem;
}

/* أو بالتفصيل */
.element {
    padding-top: 1rem;
    padding-right: 2rem;
    padding-bottom: 3rem;
    padding-left: 4rem;
}

9. margin (المسافة الخارجية)

margin: 5px auto; → أعلى/أسفل 5px، يمين/يسار auto (توسيط)

                    
/* margin */
.element {
    margin: 5px auto;
}

/* أو بالتفصيل */
.element {
    margin-top: 5px;
    margin-right: auto;
    margin-bottom: 5px;
    margin-left: auto;
}
